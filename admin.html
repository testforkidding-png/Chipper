🗄 Supabase Kurulum SQL

config.js'te USE_SUPABASE=true yapıp aşağıdaki SQL'i Supabase SQL Editor'da çalıştırın:

-- USERS table
CREATE TABLE users (
  username     TEXT PRIMARY KEY,
  password_hash TEXT NOT NULL,
  display_name TEXT,
  bio          TEXT DEFAULT '',
  avatar_url   TEXT,
  banner_color TEXT DEFAULT '#0A1628',
  status       TEXT DEFAULT '',
  status_emoji TEXT DEFAULT '',
  is_admin     BOOLEAN DEFAULT false,
  badges       TEXT[] DEFAULT '{}',
  email        TEXT,
  created_at   BIGINT DEFAULT EXTRACT(EPOCH FROM NOW())*1000
);

-- CONVERSATIONS table
CREATE TABLE conversations (
  id           TEXT PRIMARY KEY,
  type         TEXT NOT NULL CHECK (type IN ('direct','group')),
  name         TEXT,
  participants TEXT[] NOT NULL,
  avatar       TEXT,
  banner_color TEXT DEFAULT '#0A1628',
  last_msg     TEXT DEFAULT '',
  last_time    BIGINT DEFAULT 0,
  unread_for   JSONB DEFAULT '{}',
  admin        TEXT,
  created_at   TIMESTAMPTZ DEFAULT NOW()
);

-- MESSAGES table
CREATE TABLE messages (
  id           TEXT PRIMARY KEY DEFAULT gen_random_uuid()::text,
  conv_id      TEXT NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
  "from"       TEXT NOT NULL,
  text         TEXT DEFAULT '',
  type         TEXT DEFAULT 'text',
  file_data    TEXT,
  file_name    TEXT,
  file_type    TEXT,
  audio_data   TEXT,
  duration     TEXT,
  transcript   TEXT,
  reply_to     TEXT,
  reply_to_text TEXT,
  reactions    JSONB DEFAULT '{}',
  status       TEXT DEFAULT 'sent',
  edited       BOOLEAN DEFAULT false,
  recalled     BOOLEAN DEFAULT false,
  destruct_at  BIGINT,
  created_at   BIGINT DEFAULT EXTRACT(EPOCH FROM NOW())*1000
);

-- STORIES table
CREATE TABLE stories (
  id         TEXT PRIMARY KEY DEFAULT gen_random_uuid()::text,
  user_id    TEXT NOT NULL,
  text       TEXT NOT NULL,
  seen_by    TEXT[] DEFAULT '{}',
  created_at TIMESTAMPTZ DEFAULT NOW(),
  expires_at TIMESTAMPTZ DEFAULT NOW() + INTERVAL '24 hours'
);

-- Row Level Security (recommended)
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Users can read all" ON users FOR SELECT USING (true);
CREATE POLICY "Only service role can write" ON users FOR ALL USING (auth.role() = 'service_role');
