// admin.html içindeki mevcut script alanını bununla güncelle const BOT_USERNAME = 'cipher_bot'; const BOT_EMOJI = { info: 'ℹ️', alert: '⚠️', gift: '🎁', system: '🤖' }; async function initAdmin() { const pass = prompt('Admin Key:'); if (pass !== CONFIG.ADMIN_KEY) { alert('Yetkisiz erişim!'); window.location.href = 'index.html'; return; } document.getElementById('admin-content').classList.remove('hidden'); await loadUsers(); } // Kullanıcıları Supabase'den çeker ve listeler async function loadUsers() { const users = await DB.getAllUsers(); const list = document.getElementById('user-list'); list.innerHTML = users.map(u => `
${u.display_name || u.username}
@${u.username} ${u.locked ? '(Kilitli)' : ''}
`).join(''); } // Yeni kullanıcı oluşturma (Artık Supabase'e yazar) async function createUser() { const u = document.getElementById('u').value.toLowerCase().trim(); const d = document.getElementById('d').value.trim(); const p = document.getElementById('p').value; if (!u || !p) return UI.toast('Eksik bilgi', 'error'); try { // Şifreyi hash'leyerek kaydet const hash = await Auth._sha256(p); await DB.createUser({ username: u, display_name: d || u, password_hash: hash, locked: false, created_at: new Date().toISOString() }); UI.toast('Kullanıcı oluşturuldu', 'success'); loadUsers(); } catch (e) { UI.toast('Hata: ' + e.message, 'error'); } } // Kullanıcıyı veritabanından tamamen siler async function deleteUser(username) { if (!confirm(`${username} silinecek?`)) return; await DB.deleteUser(username); loadUsers(); } // Sayfa açıldığında başlat window.onload = initAdmin;
