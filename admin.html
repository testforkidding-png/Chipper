3️⃣ SQL Editor → Aşağıdaki sorguyu çalıştır
-- CIPHER v6 — Supabase kurulum SQL
-- SQL Editor'da çalıştırın (mevcut tablo varsa ALTER kullanın)

CREATE TABLE IF NOT EXISTS users (
  username      TEXT PRIMARY KEY,
  password_hash TEXT NOT NULL,
  display_name  TEXT NOT NULL DEFAULT '',
  bio           TEXT DEFAULT '',
  avatar_url    TEXT,
  banner_color  TEXT DEFAULT '#0A1628',
  status        TEXT DEFAULT '',
  status_emoji  TEXT DEFAULT '',
  is_admin      BOOLEAN DEFAULT false,
  locked        BOOLEAN DEFAULT false,
  badges        TEXT[] DEFAULT '{}',
  server_roles  JSONB DEFAULT '{"friends":false,"private":false,"public":true,"family":false}',
  last_seen     BIGINT DEFAULT 0,
  online        BOOLEAN DEFAULT false,
  stale_hash    BOOLEAN DEFAULT false,
  created_at    BIGINT DEFAULT EXTRACT(EPOCH FROM NOW())*1000
);

-- Eğer tablo zaten varsa eksik kolonları ekle:
ALTER TABLE users ADD COLUMN IF NOT EXISTS server_roles JSONB DEFAULT '{"friends":false,"private":false,"public":true,"family":false}';
ALTER TABLE users ADD COLUMN IF NOT EXISTS last_seen BIGINT DEFAULT 0;
ALTER TABLE users ADD COLUMN IF NOT EXISTS online BOOLEAN DEFAULT false;
ALTER TABLE users ADD COLUMN IF NOT EXISTS stale_hash BOOLEAN DEFAULT false;

CREATE TABLE IF NOT EXISTS conversations (
  id           TEXT PRIMARY KEY,
  type         TEXT NOT NULL DEFAULT 'direct',
  name         TEXT,
  participants TEXT[] NOT NULL DEFAULT '{}',
  avatar       TEXT,
  banner_color TEXT DEFAULT '#0A1628',
  last_msg     TEXT DEFAULT '',
  last_time    BIGINT DEFAULT 0,
  last_from    TEXT DEFAULT '',
  unread_for   JSONB DEFAULT '{}',
  admin        TEXT,
  server       TEXT DEFAULT 'public',
  created_at   BIGINT DEFAULT EXTRACT(EPOCH FROM NOW())*1000
);

-- Mevcut tablo için:
ALTER TABLE conversations ADD COLUMN IF NOT EXISTS last_from TEXT DEFAULT '';
ALTER TABLE conversations ADD COLUMN IF NOT EXISTS server TEXT DEFAULT 'public';

CREATE TABLE IF NOT EXISTS messages (
  id            TEXT PRIMARY KEY DEFAULT gen_random_uuid()::text,
  conv_id       TEXT NOT NULL,
  "from"        TEXT NOT NULL,
  text          TEXT DEFAULT '',
  type          TEXT DEFAULT 'text',
  file_data     TEXT,
  file_name     TEXT,
  file_type     TEXT,
  audio_data    TEXT,
  duration      TEXT,
  gif_url       TEXT,
  gif_title     TEXT,
  sticker       TEXT,
  reply_to      TEXT,
  reply_to_text TEXT,
  reactions     JSONB DEFAULT '{}',
  status        TEXT DEFAULT 'sent',
  edited        BOOLEAN DEFAULT false,
  recalled      BOOLEAN DEFAULT false,
  destruct_at   BIGINT,
  created_at    BIGINT DEFAULT EXTRACT(EPOCH FROM NOW())*1000
);

CREATE TABLE IF NOT EXISTS stories (
  id         TEXT PRIMARY KEY DEFAULT gen_random_uuid()::text,
  user_id    TEXT NOT NULL,
  text       TEXT NOT NULL,
  seen_by    TEXT[] DEFAULT '{}',
  created_at TIMESTAMPTZ DEFAULT NOW(),
  expires_at TIMESTAMPTZ DEFAULT NOW() + INTERVAL '24 hours'
);

ALTER PUBLICATION supabase_realtime ADD TABLE messages;
ALTER PUBLICATION supabase_realtime ADD TABLE conversations;
ALTER TABLE users         DISABLE ROW LEVEL SECURITY;
ALTER TABLE conversations DISABLE ROW LEVEL SECURITY;
ALTER TABLE messages      DISABLE ROW LEVEL SECURITY;
ALTER TABLE stories       DISABLE ROW LEVEL SECURITY;
